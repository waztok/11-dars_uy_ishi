#include <stdio.h>

int main() {
    int i, t, c = 0;

    for(i = 1; i <= 7; i++) {
        scanf("%d", &t);

        if(t > 30) {
            printf("issiq\n");
            c = c + 1;
        }
        if(t >= 15 && t <= 30) {
            printf("iliq\n");
        }
        if(t < 15) {
            printf("sovuq\n");
        }
    }

    printf("%d", c);

    return 0;
}
