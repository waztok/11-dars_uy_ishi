        printf("4. hisobni ko'rish\n");

        scanf("%d", &t);

        switch(t) {
            case 1:
                scanf("%d", &m);
                s = s + m * 5000;
                break;
            case 2:
                scanf("%d", &m);
                s = s + m * 12000;
                break;
            case 3:
                scanf("%d", &m);
                s = s + m * 10000;
                break;
            case 4:
                printf("%d\n", s);
                break;
            case 5:
                printf("%d\n", s);
                break;

    }

    return 0;
}
