#include <stdio.h>

int main() {
    int n, q;
    int r = 0, j = 0, t = 0, s = 0;

    scanf("%d", &n);

    while(n > 0) {
        q = n % 10;
        r = r + 1;
        s = s + q;

        if(q % 2 == 0) {
            j = j + 1;
        }
        if(q % 2 != 0) {
            t = t + 1;
        }

        n = n / 10;
    }

    printf("%d\n", r);
    printf("%d\n", j);
    printf("%d\n", t);
    printf("%d\n", s);

    return 0;
}
